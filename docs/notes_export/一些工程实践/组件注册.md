# 组件注册

# 声明与定义

由于引擎事先不知道游戏具体会自定义哪些组件，而改造本身又平等地涉及每个组件的修改，所以引擎可以规定一套统一的组件注册入口：

```c
#define DECLARE_COMP_BEGIN(comp_name) typedef struct comp_name { Entity owner,
#define DECLARE_COMP_END(comp_name) } comp_name; \
extern comp_name* __ds_##comp_name; \
extern size_t __len_##comp_name; \
extern int* __ss_##comp_name;

#define IMPL_COMP(comp_name) \
comp_name __ds_##comp_name[ENTITY_MAX_NUM]; \
size_t __len_##comp_name = 0; \
int __ss_##comp_name[ENTITY_MAX_NUM];
```

这样组件的注册方式也得到了简化，如之前的`Position`组件：

```c
// .h
DECLARE_COMP_BEGIN(Position)
	float x;
	float y;
DECLARE_COMP_END(Position)

// .c
IMPL_COMP(Position)
```

> 这样还降低了编写出错的几率。

这样，通过新方式注册的组件，它们的结构体**自动”注入“了Entity字段**。稠密集每个元素的Entity就可以用来反向指示该组件的实体。

那么，之前涉及到组件的插入的部分，实体归属改变，故这种情况需要添加修改Entity字段的逻辑：

```c
bool ecs_ins(void* dense_set, int* dense_len, int* sparse_set, const void* comp, size_t comp_size, Entity e) {
	if (ecs_has(sparse_set, e)) return false; // 一个实体最多拥有一个同种的组件
	sparse_set[e] = dense_len;
	void* dst = dense_set + dense_len * comp_size;
	memcpy(dst, comp, comp_size);
	*((Entity*)dst) = e; // 更新所属实体字段
	dense_len++;
	return true;
}
```

这样一来，既能通过实体查询组件，也能反向通过组件查询实体，进一步查询同实体的其他组件。实现双链。

# 注册

可以考虑规定一个组件最大数量，限制注册上限：

```c
#define COMP_MAX_NUM 128
```

不过为了简便，这里省略掉。

在前面提到过*组件注册表*，这里进行一些改造，并采用stb_ds.h的哈希表的实现方式：

```c
// .h
typedef struct {
	void* dense_set;
	int* dense_len;
	int* sparse_set;
	size_t comp_size;
} CompType;
typedef struct {
	uint32_t key;
	CompType type;
} CompTypeEntry, *CompTypeReg;
typedef struct {
	const char* key;
	uint32_t id;
} CompNameEntry, *CompNameReg;

uint32_t comp_id(const char* name);
uint32_t reg_comp(const char* name, CompType type);

#define COMP_ID(comp_name) comp_id(#comp_name)
#define REG_COMP(comp_name) \
do { \
	CompType type = (CompType) { \
		__ds_##comp_name, __len_##comp_name, __ss_##comp_name, sizeof(comp_name) \
	}; \
	reg_comp(#comp_name, type); \
} while(0)

```

源文件：

```c
// .c
CompTypeReg __comp_type_reg = NULL;
CompNameReg __comp_name_reg = NULL;

// 带缓存机制的组件ID生成器
static uint32_t gen_comp_id(const char* name) {
	static uint32_t __comp_type_id_counter = 0; // 0是合法初始值
	ptrdiff_t i = shgeti(__comp_name_reg, name);
	if (i >= 0) {
		return __comp_name_reg[i].id;
	}
	shput(__comp_name_reg, name, __comp_type_id_counter);
	return __comp_type_id_counter++; // 如果有上限限制，加超出上限判断，若超出，返回非法值UINT_MAX(<limits.h>)
}

// 组件ID获取器
uint32_t comp_id(const char* name) {
	ptrdiff_t i = shgeti(__comp_name_reg, name);
	if (i >= 0) {
		return __comp_name_reg[i].id;
	}
	return UINT_MAX; // 查不到说明组件未注册，返回非法值UINT_MAX
}

// 带缓存机制的组件注册器
uint32_t reg_comp(const char* name, CompType type) {
	uint32_t id = gen_comp_id(name);
	ptrdiff_t i = hmgeti(__comp_type_reg, id);
	if (i < 0) {
		hmput(__comp_type_reg, id, type);
	}
	return id;
}
```

这样，使用它只需在main的开头调用一下：

```c
#include ...
int main(void) {
	// 注册begin
	REG_COMP(Position);
	REG_COMP(Velocity);
	...
	// 注册end
	... // 游戏过程中任何地方需要获取组件ID：例如 COMP_ID(Position) 这样即可
	return 0;
}
```

到这里，无论是组件的声明、定义还是注册，其API都变得十分友好。参数部分都**只需传递一个组件类型标识符**。
