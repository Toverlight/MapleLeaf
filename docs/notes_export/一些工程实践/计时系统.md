# 计时系统

# 需要被更新的时间资源

主要是两类，一个是**TimeFixed**，用于存储游戏每帧期望时间（固定帧间隔），全局固定不更改，提供三个常用计量单位：ns, ms, s；另一个是**TimeDelta**，用于存储上一帧到当前帧的耗时，每帧更新一次，同样三个计量单位。

```c
// .h
DECLARE_RESOURCE_BEGIN()
	uint64_t nspf;
	double mspf;
	double spf;
DECLARE_RESOURCE_END(Res_TimeFixed)

DECLARE_RESOURCE_BEGIN()
	uint64_t dns;
	double dms;
	double ds;
DECLARE_RESOURCE_END(Res_TimeDelta)

// .c
IMPL_RESOURCE(Res_TimeFixed)
IMPL_RESOURCE(Res_TimeDelta)
```

然后还有在程序入口内置相关资源注册语句。

# 获取时间资源的接口

时间资源在一般游戏中会经常用到，用户需要编写大量相关**获取**代码。所以最好对已有获取方式`GET_RES()`进行一个简化封装。

**获取Fixed Time：**

```c
#define get_fixed_ns() GET_RES(Res_TimeFixed, nspf)
#define get_fixed_ms() GET_RES(Res_TimeFixed, mspf)
#define get_fixed_s() GET_RES(Res_TimeFixed, spf)
```

**获取Delta Time：**

```c
#define get_delta_ns() GET_RES(Res_TimeDelta, dns)
#define get_delta_ms() GET_RES(Res_TimeDelta, dms)
#define get_delta_s() GET_RES(Res_TimeDelta, ds)
```

**除此之外，还应有获取当前时间的接口：**

```c
#define get_current_ns() ...
#define get_current_ms() ...
#define get_current_s() ...
```

# 计时器

**计时器（Timer）** 具有如下三种功能：

- **测量时间**：记录开始到结束的时间。
- **停表**：开始后，用户随时暂停，并获取度过的时间。
- **定时任务**：设定间隔和回调函数，每度过固定间隔就触发一次回调，可以设定触发次数。

计时器启动(Start)时，记录（更新）开始时间；中途可以暂停(Pause)和继续(Resume)，改变“运行中”状态；"运行中"且触发次数不等于0时，计时器允许被更新；更新后到点（最新时间 - 开始时间 >= 预设间隔）时，触发一次回调（如果有），然后减少一次触发次数（如果为正）并更新开始时间为最新时间。

```c
// .h
struct Timer;
typedef void(*TimerCallback)(const struct Timer* self, uint64_t* interval_ns, void* data);
typedef struct Timer {
	bool running; // “运行中”状态
	int rest_count; // 剩余触发次数。小于0时是无限，0时停止，大于0时预期还会触发该次数
	uint64_t interval_ns; // 设置的间隔。值可以很小（到0），此时每帧都触发
	uint64_t start_timepoint_ns; // “启动”或“触发”时更新
	double start_timepoint_ms; // ms等价辅助字段
	double start_timepoint_s; // s等价辅助字段
	uint64_t latest_timepoint_ns; // 每次更新计时器时更新
	double latest_timepoint_ms; // ms等价辅助字段
	double latest_timepoint_s; // s等价辅助字段
	void* data; // 用户自定义数据
	TimerCallback callback;
} Timer;

#define TIMER_INFINITY -1
#define TIMER_STOP 0

Timer* timer_new(const char* name, int rest_count, uint64_t interval_ns, void* data, TimerCallback callback);
Timer* timer_get_or_null(const char* name);
void timer_set_rest_count(Timer* timer);
void timer_start(Timer* timer);
void timer_pause(Timer* timer);
void timer_resume(Timer* timer);
void timer_stop(Timer* timer);
bool timer_delete(const char* name);
```

所有的timer，在用户自定义的系统内创建，通过`timer_new()`函数。该函数内部在堆区创建timer后，将其地址注册到哈希表中，并返回该地址。

```c
// .c
typedef struct {
	const char* name;
	Timer* timer;
} TimerEntry;

static TimerEntry* timer_reg = NULL;

Timer* timer_new_fn(const char* name, int rest_count, uint64_t interval_ns, void* data, TimerCallback callback) {
	Timer* timer = (Timer*)malloc(sizeof(Timer));
	timer.running = false;
	timer.rest_count = rest_count;
	timer.interval_ns = interval_ns;
	timer.data = data;
	timer.callback = callback;

	shput(timer_reg, name, timer);
}

Timer* timer_get_or_null_fn(const char* name) {
	ptrdiff_t i = shgeti(timer_reg, name);
	if (i >= 0) {
		return timer_reg[i].timer;	
	}
	return NULL;
}

void timer_set_rest_count_fn(Timer* timer, int rest_count) {
	if (timer) timer.rest_count = rest_count;
}

void timer_start_fn(Timer* timer) {
	if (!timer) return;
	timer.running = true;
	timer.start_timepoint_ns = get_current_ns();
	timer.start_timepoint_ms = (double)timer.start_timepoint_ns / RATIO_MS_NS;
	timer.start_timepoint_s = (double)timer.start_timepoint_ns / RATIO_S_NS;
}

void timer_pause_fn(Timer* timer) {
	if (timer) timer.running = false;
}

void timer_resume_fn(Timer* timer) {
	if (timer) timer.running = true;
}

void timer_stop_fn(Timer* timer) {
	if (!timer) return;
	timer.running = false;
	timer.rest_count = 0;
}

bool timer_delete_fn(const char* name) {
	ptrdiff_t i = shgeti(timer_reg, name);
	if (i >= 0) {
		free(timer_reg[i].timer);
		shdel(timer_reg, name);
		return true;
	}
	return false;
}
```

> 程序出口处记得将`timer_reg`中所有项的timer指针释放。

接下来是负责更新计时器的函数。

```c
// .h
void timer_update_fn(Timer* timer);

// .c
void timer_update_fn(Timer* timer) {
	if (!timer) return;
	if (timer.running && timer.rest_count != TIMER_STOP) {
		timer.latest_timepoint_ns = get_current_ns();
		timer.latest_timepoint_ms = (double)timer.latest_timepoint_ns / RATIO_MS_NS;
		timer.latest_timepoint_s = (double)timer.latest_timepoint_ns / RATIO_S_NS;
		if (timer.latest_timepoint_ns - timer.start_timepoint_ns >= timer.interval_ns) {
			if (timer.callback) timer.callback(timer, &timer.interval_ns, timer.data);
			if (timer.rest_count > 0) timer.rest_count--;
			timer.start_timepoint_ns = get_current_ns();
			timer.start_timepoint_ms = (double)timer.start_timepoint_ns / RATIO_MS_NS;
			timer.start_timepoint_s = (double)timer.start_timepoint_ns / RATIO_S_NS;
		}
	}
}
```

为了计时器更新的准确性，这里没有采用统一“最新”时间记录而是采用各存一份、独立更新的方式。

接下来，仅需在系统内调用一次`timer_update()`函数即可让计时器活跃。
