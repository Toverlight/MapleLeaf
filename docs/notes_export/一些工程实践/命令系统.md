# 命令系统

实体的生成与销毁、组件的添加与删除、实体关系的构筑等，都需要频繁地用到，这些操作被统筹为**命令**。

命令分为**实体无关**和**实体相关**两种。分别叫作Command和EntityCommand。

两者的区别在于：EntityCommand实例会保存实体，内部函数操作与该实体紧密相关；Command通常经内部函数操作产生EntityCommand。

# 实体与组件的增删

为了方便编程，命令被设计成可被链式调用的形式：

```c
// .h
// 'cmd' prefix
typedef struct Command {
	// ...
} Command;
// 'ecmd' prefix
typedef struct EntityCommand {
	Entity target;
	struct Command* cmd;
} EntityCommand;

extern Command command; // 暂时用单例。后面会优化

#define COMMAND() &(command)

// .c
Command command;
```

以下接口包括了实体和组件的增删。

void ecmd_insert_fn(struct EntityCommand* self, uint32_t comp_id, const void* comp_data, bool* success) {
	if (!self->target) {
		if (success) *success = false;
		return;
	}
	ptrdiff_t i = hmgeti(__comp_type_reg, comp_id);
	if (i >= 0) {
		CompType type = __comp_type_reg[i].type;
		if (ecs_ins(type, comp_data, self->target))
			if (success) *success = true;
		else
			if (success) *success = false;
	} else {
		if (success) *success = false;
	}
	return;
}
void ecmd_remove_fn(struct EntityCommand* self, uint32_t comp_id, void* out, bool* success) {
	if (!self->target) {
		if (success) *success = false;
		return;
	}
	ptrdiff_t i = hmgeti(__comp_type_reg, comp_id);
	if (i >= 0) {
		CompType type = __comp_type_reg[i].type;
		if (ecs_del(type, self->target, out))
			if (success) *success = true;
		else
			if (success) *success = false;
	} else {
		if (success) *success = false;
	}
	return;
}
void ecmd_despawn_fn(struct EntityCommand* self, bool* success) {
	if (!self->target) {
		if (success) *success = false;
	}
	if (despawn_entity(self->target)) 
		if (success) *success = true;
	else
		if (success) *success = false;
}

struct EntityCommand cmd_spawn_fn(struct Command* self, bool* success) {
	Entity target = next_entity();
	EntityCommand ecmd = { target, self };
	if (!target) {
		// 加一条限定最大次数的警告或报错，因为这种情况会导致后面的链式调用全部失败
		if (success) *success = false;
	} else {

	return ecmd;
}
struct EntityCommand cmd_entity_fn(struct Command* self, Entity target, bool* success) {
	EntityCommand ecmd = { target, self };
	if (!target) {
		// 加一条限定最大次数的警告或报错，因为这种情况会导致后面的链式调用全部失败
		if (success) *success = false;
	} else {
		if (success) *success = true;
	}
	return ecmd;
}

# 关系依赖

实体之间可能存在依赖关系。最常见的关系是**父子关系**。

父子关系的特征是**子实体依赖父实体**。所以只需建立一个子到父的映射即可表示。

但是，很多时候我们有获取一个实体的子实体的需求（比如级联删除），所以，对于每一个作为“键”的实体，应该对应兼具**指上和指下**的结构体：其中父实体只需一个实体id，而子实体需要指向一个动态数组。

## 结构定义

这里采用哈希表：

```c
// .h
typedef struct {
	Entity parent;
	Entity* children;
} Filiation;
typedef struct {
	Entity key;
	Filiation filiation;
} FiliationEntry, *FiliationReg;

// .c
FiliationReg filiation_reg = NULL;
```

## 构建时添加依赖

添加时，**同时确保该实体键和其父/子实体（“被关联”实体）键都存在**，并且：

- 若**被关联为父**：该实体的`filiation`​的`parent`​指向父，父的`filiation`​的`children`添加该实体。
- 若**被关联为子**：该实体的`filiation`​的`children`​添加子，父的`filiation`​的`parent`指向该实体。

```c
// .h
void ecmd_with_child_fn(struct EntityCommand* self, Entity child, bool* success);
#define ecmd_with_child(success, ...) \
do { \
	EntityCommand* _obj_parent = _obj_self; \
	EntityCommand _obj_ecmd = cmd_spawn(_obj_parent->cmd, (success)); \
	EntityCommand* _obj_self = &_obj_ecmd; \
	__VA_ARGS__; \
	ecmd_with_child_fn(_obj_parent, _obj_self->target, (success)); \
} while(0)

// .c
void ecmd_with_child_fn(struct EntityCommand* self, Entity child, bool* success) {
	if (!self->target || !child) {
		if (success) *success = false;
		return;
	}
	// 添加子
	ptrdiff_t p_i = hmgeti(filiation_reg, self.target);
	if (p_i < 0) {
		Filiation filiation = { 0, NULL };
		arrput(filiation.children, child);
		hmput(filiation_reg, self.target, filiation);
	} else {
		arrput(filiation_reg[p_i].filiation.children, child);
	}
	// 指向父
	ptrdiff_t c_i = hmgeti(filiation_reg, self.target);
	if (c_i < 0) {
		Filiation filiation = { self.target, NULL };
		hmput(filiation_reg, child, filiation);
	} else {
		filiation_reg[c_i].filiation.parent = self.target;
	}
	if (success) *success = true;
}
```

**使用：**

```c
EntityCommand ecmd = cmd_spawn(COMMAND(), NULL);
WITH(ecmd, // 注意，第一个用逗号
	// 之后的用分号隔开
	ecmd_insert(Position, data_pos, NULL);
	ecmd_insert(HealthPoint, data_hp, NULL);
	ecmd_with_child(NULL, // 逗号
		ecmd_insert(Position, data_child_pos, NULL);
		ecmd_with_child(NULL, // 逗号
			ecmd_insert(Something, data_gc_sth, NULL);
		);
	);	
);
```

## 动态换绑

换绑，就是实体更换挂载到的父实体。

```c
// 检查filiation存在必要性，即若无parent且children为空，则该项无存在必要，删除该项
void myproj_entity_test_filiation_fn(Entity self) {
	ptrdiff_t i = hmgeti(filiation_reg, self);
	if (i >= 0) {
		if (!filiation_reg[i].filiation.parent 
			&& !arrlen(filiation_reg[i].filiation.children))
		{
			arrfree(filiation_reg[i].children);
			hmdel(filiation_reg, self);
		}
	}
}
void myproj_entity_set_parent_fn(Entity self, Entity parent) {
	if (!self) return;
	// self项
	ptrdiff_t i = hmgeti(filiation_reg, self);
	if (i >= 0) {
	Proc:
		// par_old项
		Entity par_old = filiation_reg[i].filiation.parent;
		if (par_old == parent) return;
		if (par_old) {
			ptrdiff_t j = hmgeti(filiation_reg, par_old);
			if (j >= 0) {
				for (size_t i = 0; i < arrlen(filiation_reg[j].filiation.children); i++) {
					if (filiation_reg[j].filiation.children[i] == self) {
						arrdel(filiation_reg[j].filiation.children, i);
						break;
					}
				}
				myproj_entity_test_filiation_fn(par_old);
			}
		}
		// parent项
		filiation_reg[i].filiation.parent = parent;
		myproj_entity_test_filiation_fn(self);
		if (parent) {
			ptrdiff_t k = hmgeti(filiation_reg, parent);
			if (k >= 0) {
				bool existent = false;
				for (size_t i = 0; i < arrlen(filiation_reg[k].filiation.children); i++) {
					if (filiation_reg[k].filiation.children[i] == self) {
						existent = true;
						break;
					}
				}
				if (!existent)
					arrput(filiation_reg[k].filiation.children, self);
			}
		}
	} else {
		Filiation filiation = { 0, NULL };
		hmput(filiation_reg, self, filiation);
		goto Proc;
	}
}
```

## 级联销毁

级联删除，意味着以当前实体为根，自上而下删除依赖树中的所有实体。

**Despawn算法：**

当前实体若有父实体，从其父实体的`children`列表中删除。

接下来准备一个实体id的暂存数组`arr`​(=NULL)和索引`cur`​(=0)。首先将当前实体加入`arr`。从当前实体开始：

1. 对该实体调用**单体删除**操作。
2. 将子实体id全部加入到`arr`。
3. 释放`children`数组。
4. 删除该实体键。
5. `cur`​++，**若未越界**，对实体`arr[cur]`进行(1~5)相同处理。

最后释放`arr`。

为确保用户调用的Despawn操作是级联的而不是单体的，建议将原来的`entity_despawn()`​（作用于单体）[重命名](siyuan://blocks/20260801231651-q1bujkh)为带项目名前缀的表示**语义上的“私有”** ，如`myproj_entity_despawn()`。

用户调用Despawn通过命令接口的`ecmd_despawn()`​，其内部应用这一算法，内部调用的单体删除操作就是`myproj_entity_despawn()`。

**实现（对之前的**​**​`ecmd_despawn_fn()`​** ​**修改）：**

```c
void ecmd_despawn_fn(struct EntityCommand* self, bool* success) {
	if (!self->target) {
		if (success) *success = false;
	}
	// 初始化arr
	Entity* arr = NULL;
	size_t cur = 0;
	myproj_entity_set_parent_fn(self.target, 0);
	arrput(arr, self.target);
	bool failed = false;
	do {
		// 单体删除
		if (!myproj_entity_despawn(arr[cur])) {
			failed = true;
			// 建议插入一条Log error指示哪个实体despawn出错
		}
		ptrdiff_t i = hmgeti(filiation_reg, arr[cur]);
		if (i >= 0) {
			// 添加子实体id到arr
			for (int j = 0; j < arrlen(filiation_reg[i].filiation.children); j++) {
				arrput(arr, filiation_reg[i].filiation.children[j]);
			}
			// 释放children
			arrfree(filiation_reg[i].filiation.children);
			// 删除该实体键
			hmdel(filiation_reg, self.target);
		}
		cur++;
	} while (cur < arrlen(arr));
	// 释放arr
	arrfree(arr);

	if (failed) 
		if (success) *success = false;
	else
		if (success) *success = true;
}
```

> 该算法是迭代的，效率高。
